[ASSISTANT] Your login, registration, database access and ownership checks have to stay in PHP. The best candidates for JS are the interactions that currently reload the whole page.



## Good candidates



1. **Toggling a task done** (`toggle_task.php`, the checkbox with `onchange="this.form.submit()"`). This is the best first conversion. Checking a box reloads the page and loses your scroll position. With JS you'd `fetch` a POST, then update the row's `todo-item--done` class and the counts. PHP would still do the DB update, but it would return JSON or a status code instead of redirecting.



2. **Deleting a task** (`delete_task.php`, the `confirm()` link). Right now it's a GET link that changes data, which isn't safe because a prefetch or a pasted link can delete tasks. JS lets you send a `DELETE` or POST with `fetch` and remove the `<li>` from the page.



3. **The "Add New Task" form** (`tasks.php?action=add`). Opening the form currently reloads the page with a query string. JS could show or hide the partial instead. You could also submit it with `fetch` and add the new row without reloading.



4. **Client-side form validation** (`create_task.html.twig`, the login and register forms). Checks like required fields and date format would give instant feedback. This only adds convenience. Keep the PHP validation in `save_task.php`, because users can bypass JS.



5. **Filter tabs** (`tasks.php?status=`). This is lower priority. Each tab is a full page load, which works fine. You could filter the already-loaded list in JS, but then the counts and the filtering logic exist in two places.



## Keep in PHP



- **Authentication and sessions**: `login_process.php`, `register_process.php` and `logout.php`. Anything that handles passwords or trust belongs on the server.

- **Authorization**: every `user_id` check comes from the session. Even with JS, each endpoint must still verify ownership.

- **Initial rendering with Twig**: the first page load should still be server-rendered. That is also what you just converted.

- **Server-side validation and whitelists**: the priority and status whitelists and the date check in `save_task.php`.

// Handles the task list in the browser so the page does not reload.

// The server (PHP) is only called to save a change in the database.



var statusLabels = {

    not_started: 'Not Started',

    in_progress: 'In Progress',

    completed: 'Completed'

};



// Show or hide the "Add New Task" form

document.getElementById('add-task-btn').addEventListener('click', function () {

    var form = document.getElementById('task-form');

    if (form.style.display === 'none') {

        form.style.display = 'block';

    } else {

        form.style.display = 'none';

    }

});



// Tick or untick a task

function toggleTask(item) {

    var checkbox = item.querySelector('.todo-item_checkbox');



    var data = new FormData();

    data.append('task_id', item.dataset.id);

    if (checkbox.checked) {

        data.append('is_done', '1');

    }



    fetch('toggle_task.php', { method: 'POST', body: data })

        .then(function (response) { return response.text(); })

        .then(function (text) {

            if (text !== 'ok') {

                checkbox.checked = !checkbox.checked; // save failed, undo

                return;

            }

            item.classList.toggle('todo-item--done', checkbox.checked);



            // Only the Tasks page has statuses and filter counts

            if (item.dataset.status) {

                item.dataset.status = checkbox.checked ? 'completed' : 'not_started';

                item.querySelector('.badge--status').textContent = statusLabels[item.dataset.status];

                updateCounts();

                applyFilter();

            }

        });

}



// Delete a task

function deleteTask(item) {

    if (!confirm('Delete this task?')) {

        return;

    }



    var data = new FormData();

    data.append('id', item.dataset.id);



    fetch('delete_task.php', { method: 'POST', body: data })

        .then(function (response) { return response.text(); })

        .then(function (text) {

            if (text === 'ok') {

                item.remove();

                updateCounts();

                applyFilter();

            }

        });

}



// One listener on the list handles every row, even rows added later

document.addEventListener('change', function (event) {

    if (event.target.classList.contains('todo-item_checkbox')) {

        toggleTask(event.target.closest('.todo-item'));

    }

});



document.addEventListener('click', function (event) {

    if (event.target.classList.contains('btn-delete')) {

        event.preventDefault();

        deleteTask(event.target.closest('.todo-item'));

    }

    if (event.target.dataset.filter) {

        event.preventDefault();

        currentFilter = event.target.dataset.filter;

        applyFilter();

    }

});



// Filter tabs (Tasks page only)

var currentFilter = 'all';



function applyFilter() {

    var items = document.querySelectorAll('.todo-item');

    var shown = 0;



    items.forEach(function (item) {

        var visible = currentFilter === 'all' || item.dataset.status === currentFilter;

        item.style.display = visible ? '' : 'none';

        if (visible) {

            shown++;

        }

    });



    document.querySelectorAll('.task-filter').forEach(function (tab) {

        tab.classList.toggle('task-filter--active', tab.dataset.filter === currentFilter);

    });



    document.getElementById('empty-message').style.display = shown === 0 ? 'block' : 'none';

}



function updateCounts() {

    var counts = { all: 0, not_started: 0, in_progress: 0, completed: 0 };



    document.querySelectorAll('.todo-item').forEach(function (item) {

        counts.all++;

        counts[item.dataset.status]++;

    });



    document.querySelectorAll('[data-count]').forEach(function (span) {

        span.textContent = counts[span.dataset.count];

    });

}



if (document.querySelector('.task-filters')) {

    updateCounts();

}

EOF
